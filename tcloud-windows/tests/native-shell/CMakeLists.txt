cmake_minimum_required(VERSION 3.24)
project(TCloudSyntheticShell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(TCloud.ShellProbe SHARED probe-folder.cpp exports.def)
add_library(TCloud.ShellExtension SHARED ../../src/TCloud.ShellExtension/folder.cpp exports.def)
add_executable(TCloud.ShellViewHost WIN32 view-host.cpp)
foreach(target TCloud.ShellProbe TCloud.ShellExtension TCloud.ShellViewHost)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
  target_link_libraries(${target} PRIVATE shell32 shlwapi ole32 uuid user32 comctl32)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${TCLOUD_OUTPUT}")
endforeach()
